### 1. Database Schema Changes

**File:** `src-tauri/src/lib.rs` (migration logic, ~line 628)

**New Tables:**

```sql
-- Profile metadata (icons, favorites, display order)
CREATE TABLE profile_metadata (
    profile_id TEXT PRIMARY KEY,
    icon TEXT,                        -- Lucide icon name (e.g., "server")
    is_favorite BOOLEAN DEFAULT 0,
    display_order INTEGER DEFAULT 0,  -- For future manual sorting
    FOREIGN KEY (profile_id) REFERENCES profiles(id) ON DELETE CASCADE
);

-- Tag definitions
CREATE TABLE tags (
    id TEXT PRIMARY KEY,              -- UUID
    name TEXT NOT NULL UNIQUE,        -- Case-sensitive, max 32 chars
    color TEXT NOT NULL,              -- Hex color (#RRGGBB)
    created_at TEXT NOT NULL
);

-- Profile-tag assignments (many-to-many)
CREATE TABLE profile_tags (
    profile_id TEXT NOT NULL,
    tag_id TEXT NOT NULL,
    PRIMARY KEY (profile_id, tag_id),
    FOREIGN KEY (profile_id) REFERENCES profiles(id) ON DELETE CASCADE,
    FOREIGN KEY (tag_id) REFERENCES tags(id) ON DELETE CASCADE
);

-- Indexes for performance
CREATE INDEX idx_profile_metadata_favorite ON profile_metadata(is_favorite);
CREATE INDEX idx_profile_tags_profile ON profile_tags(profile_id);
CREATE INDEX idx_profile_tags_tag ON profile_tags(tag_id);
```

**Migration Steps:**
1. Create three new tables
2. No data migration needed (all profiles start with no metadata/tags)

```javascript
let selectedIcon = null;
let currentProfileIdForIcon = null;

function openIconPicker(profileId, currentIcon) {
    currentProfileIdForIcon = profileId;
    selectedIcon = currentIcon || null;

    renderIconGrid();
    iconPickerModal.classList.remove('hidden');
    pushModal('iconPickerModal');

    // Focus search input
    iconSearchInput.focus();
}

function closeIconPicker() {
    iconPickerModal.classList.add('hidden');
    popModal('iconPickerModal');
    currentProfileIdForIcon = null;
    selectedIcon = null;
    iconSearchInput.value = '';
}

function renderIconGrid(filterText = '') {
    const grid = document.getElementById('iconGrid');
    grid.innerHTML = '';

    const filteredIcons = PROFILE_ICONS.filter(icon =>
        icon.toLowerCase().includes(filterText.toLowerCase())
    );

    filteredIcons.forEach(iconName => {
        const item = document.createElement('div');
        item.className = 'icon-grid-item';
        if (selectedIcon === iconName) {
            item.classList.add('selected');
        }

        item.innerHTML = `
            <i data-lucide="${iconName}"></i>
            <span>${iconName}</span>
        `;

        item.addEventListener('click', () => {
            selectIcon(iconName);
        });

        grid.appendChild(item);
    });

    // Re-initialize Lucide icons for newly added elements
    lucide.createIcons();
}

function selectIcon(iconName) {
    selectedIcon = iconName;

    // Update profile icon via backend
    invoke('update_profile_icon', {
        profileId: currentProfileIdForIcon,
        icon: iconName
    }).then(() => {
        // Update local profiles array
        const profile = profiles.find(p => p.id === currentProfileIdForIcon);
        if (profile) {
            profile.icon = iconName;
        }

        renderProfiles();
        closeIconPicker();
    }).catch(err => {
        showNotification(`Failed to update icon: ${err}`, 'error');
    });
}

function clearProfileIcon() {
    invoke('update_profile_icon', {
        profileId: currentProfileIdForIcon,
        icon: null
    }).then(() => {
        // Update local profiles array
        const profile = profiles.find(p => p.id === currentProfileIdForIcon);
        if (profile) {
            profile.icon = null;
        }

        renderProfiles();
        closeIconPicker();
    }).catch(err => {
        showNotification(`Failed to clear icon: ${err}`, 'error');
    });
}

// Event listeners
iconSearchInput.addEventListener('input', (e) => {
    renderIconGrid(e.target.value);
});

clearIconButton.addEventListener('click', clearProfileIcon);
cancelIconPickerButton.addEventListener('click', closeIconPicker);
```

async function toggleProfileFavorite(profileId, event) {
    event.stopPropagation(); // Prevent card click events

    const profile = profiles.find(p => p.id === profileId);
    if (!profile) return;

    const newState = !profile.is_favorite;

    try {
        await invoke('toggle_profile_favorite', { profileId, newState });

        // Update local state
        profile.is_favorite = newState;

        // Show toast notification
        showNotification(
            newState ? 'Added to Favourites' : 'Removed from Favourites',
            'success'
        );

        // Re-render profiles
        renderProfiles();
    } catch (err) {
        showNotification(`Failed to update favourite: ${err}`, 'error');
    }
}

function parseSearchQuery(query) {
    const tagPattern = /tag:(\S+)/g;
    const tags = [];
    let textQuery = query;

    // Extract all tag: queries
    let match;
    while ((match = tagPattern.exec(query)) !== null) {
        tags.push(match[1].toLowerCase());
        textQuery = textQuery.replace(match[0], '').trim();
    }

    return {
        text: textQuery,
        tags: tags
    };
}

function handleSearch() {
    const searchInput = document.getElementById('searchInput');
    const rawQuery = searchInput.value.trim();

    const { text, tags } = parseSearchQuery(rawQuery);

    searchQuery = text;

    renderProfiles();
}
